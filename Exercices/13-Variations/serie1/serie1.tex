\begin{Answer}[ref=exe:10]
	On résoud $3x-2 = 0$ pour trouver où la fonction s'annule exactement.
		\begin{align*}
			3x - 2 &= 0 \\
			3x &= 2 \\
			x &= \frac23
		\end{align*}
		
	\begin{center}
	\begin{tikzpicture}[>=stealth]
		\begin{axis}[xmin = -4, xmax=4, ymin=-10, ymax=10, axis x line=middle, axis y line=middle, axis line style=->, grid=both,
			grid style = {opacity=.5},
			x=20pt,
			xtick distance=1,
			ytick distance=2,
			%y=20pt,
			clip=true,
			]
			\addplot[no marks, BLUE_E, -, very thick] expression[domain=-4:4, samples=2]{3*x-2}
			node[left, pos=.9]{$y=3x-2$};
		\end{axis}
	\end{tikzpicture}
	
	\begin{tikzpicture}
		\tkzTabInit
		 %[lgt=3,espcl=1.5]
	       		{$x$ / 1 , Variation de $3x-2$ / 2, Signe de $3x-2$ / 1}
	       		{  $-10$ ,,  $\dfrac23$ ,,  $12$ }

			{-/$-32$, R/, R/, R/, +/$34$}
		\tkzTabIma
			{1}{5}{3}{0}
		\tkzTabLine
			{,,-,,z,,+}
	\end{tikzpicture}
	\end{center}
\end{Answer}
